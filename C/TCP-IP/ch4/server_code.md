# 대략적인 서버 구조
socket()    // 소켓 생성
bind()      // 소켓에 주소 할당
listen()    // 연결 요청 대기 상태
accept()    // 연결 허가
read/write()     // 데이터 송수신
close()          // 연결 종료
# 대략적인 클라이언 구조
socket()     // 소켓 생성
connect()    // 연결 요청
read/write()      // 데이터 송수신
close()           // 연결 종료

# iterative 서버의 구현
- accept 함수를 호출하고 데이터 송수신 이후 클라이언트를 닫고 다시 이 과정을 반복하는 서버
-accept 함수의 호출 과정에서 생성된 소켓에 대상
- 또 다른 클라이언트에게 서비스 하기 위해선 다시 호출 해야함
